CREATE TABLE #AuditTemp3
(
    [ServerName]          varchar(max),
    [LoginType]           varchar(max),
    [LoginName]           varchar(max),
    [DatabaseUserName]    varchar(max),
    [PermissionType]      varchar(max),
    [DatabaseName]        varchar(max),
    [Role]                varchar(max),
    [ReportDate]          varchar(max)
);

INSERT INTO #AuditTemp3
SELECT
    [ServerName],
    [LoginType],
    [LoginName],
    [DatabaseUserName],
    [PermissionType],
    [DatabaseName],
    [Role],
    CONVERT(varchar(20), GETDATE(), 120)
FROM [bib_dba_utilities].[dbo].[permission_audit_temp]
WHERE PermissionType LIKE 'EXPLICIT DATABASE LEVEL%'
  AND LoginType != 'Database ROLE'
  AND LoginType != 'User mapped to a CERTIFICATE'
  AND LoginName != 'INFORMATION_SCHEMA'
  AND LoginName NOT IN ('dbo','sa','guest','sys','117968-a1','117962-a1','idera_dm_admin','BIBERK\SVC_SQL_IDERA_DM','cms_admin')
  AND LoginName NOT LIKE '##%'
  AND LoginName NOT LIKE 'NT%\%'
ORDER BY LoginName, PermissionType;

DECLARE @i INT = 0, @pos INT, @s VARCHAR(MAX) = '', @ts VARCHAR(MAX), @xml VARCHAR(MAX);

SET @xml = CAST((
    SELECT
        ROW_NUMBER() OVER (ORDER BY LoginName, PermissionType) AS 'td', '',
        LoginName AS 'td', '',
        DatabaseUserName AS 'td', '',
        ('Database Level | Server: ' + ServerName + 
         ' | Database: ' + ISNULL(DatabaseName,'') + 
         ' | Role: ' + ISNULL([Role],'') + 
         ' | Permission: ' + ISNULL(PermissionType,'')) AS 'td', '',
        '' AS 'td', '',
        '' AS 'td', '',
        CASE WHEN LoginType LIKE '%GROUP%' 
             THEN 'zr' + CAST(LoginType AS VARCHAR(255))
             ELSE CAST(LoginType AS VARCHAR(255))
        END AS 'td', '',
        CASE WHEN LoginName LIKE 'SVC[_]%' OR LoginName LIKE '%[_]SVC[_]%' OR LoginName LIKE 'svc%'
             THEN 'Managed service account – see purpose column'
             ELSE 'N/A for named users'
        END AS 'td', '',
        CASE WHEN LoginName LIKE 'SVC[_]%' OR LoginName LIKE '%[_]SVC[_]%' OR LoginName LIKE 'svc%'
             THEN 'Service / application account (review purpose with owner)'
             ELSE 'N/A for named users'
        END AS 'td', '',
        CASE WHEN LoginName LIKE 'SVC[_]%' OR LoginName LIKE '%[_]SVC[_]%' OR LoginName LIKE 'svc%'
             THEN 'Confirm authorized users / process owners'
             ELSE 'N/A for named users'
        END AS 'td', '',
        '' AS 'td', '',
        '' AS 'td', '',
        '' AS 'td', '',
        '' AS 'td', '',
        '' AS 'td'
    FROM #AuditTemp3
    ORDER BY LoginName, PermissionType
    FOR XML PATH('tr'), ELEMENTS
) AS NVARCHAR(MAX));

SET @xml = REPLACE(@xml, '<td>zr', '<td id="r">');

SELECT @s = '', @pos = CHARINDEX('<tr>', @xml, 4);
WHILE (@pos > 0)
BEGIN
    SET @i += 1;
    SET @ts = SUBSTRING(@xml, 1, @pos - 1);
    IF (@i % 2 = 1) SET @ts = REPLACE(@ts, '<tr>', '<tr id="odd">');
    SET @s += @ts;
    SET @xml = SUBSTRING(@xml, @pos, LEN(@xml));
    SET @pos = CHARINDEX('<tr>', @xml, 4);
END
SET @i += 1;
SET @ts = @xml;
IF (@i % 2 = 1) SET @ts = REPLACE(@ts, '<tr>', '<tr id="odd">');
SET @s += @ts;

SET @body += @s + '</table>
<br><br>
Regards,<br>
BIBERK Database Solutions Team
</body></html>';

    @body         = @body,
