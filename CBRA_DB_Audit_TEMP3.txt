CREATE TABLE #AuditTemp3
(
    [ServerName]       varchar(max),
    [LoginType]        varchar(max),
    [LoginName]        varchar(max),
    [DatabaseUserName] varchar(max),
    [PermissionType]   varchar(max),
    [DatabaseName]     varchar(max),
    [Role]             varchar(max),
    [AccountStatus]    varchar(max),
    [ReportDate]       varchar(max)
);

INSERT INTO #AuditTemp3
SELECT
    [ServerName],
    [LoginType],
    [LoginName],
    [DatabaseUserName],
    [PermissionType],
    [DatabaseName],
    [Role],
    ISNULL([AccountStatus], 'N/A'),
    CONVERT(varchar(20), GETDATE(), 120)
FROM [bib_dba_utilities].[dbo].[permission_audit_temp]
WHERE PermissionType LIKE 'EXPLICIT DATABASE LEVEL%'
  AND LoginType != 'Database ROLE'
  AND LoginType != 'User mapped to a CERTIFICATE'
  AND LoginName != 'INFORMATION_SCHEMA'
  AND LoginName NOT IN ('dbo','sa','guest','sys','117968-a1','117962-a1','idera_dm_admin','BIBERK\SVC_SQL_IDERA_DM','cms_admin')
  AND LoginName NOT LIKE '##%'
  AND LoginName NOT LIKE 'NT%\%'
ORDER BY LoginName, PermissionType;

DECLARE @RowCount int = (SELECT COUNT(*) FROM #AuditTemp3);

DECLARE @body VARCHAR(MAX) = '
<html>
<head>
<style>
  #g {color: green;}
  #r {color: red;}
  #odd {background-color: #f2f2f2;}
  th {background-color: #4472C4; color: white; font-weight: bold; padding: 6px; border: 1px solid #999; text-align: center; vertical-align: middle; font-size: 11px;}
  td {padding: 4px 6px; border: 1px solid #999; vertical-align: top; font-size: 11px;}
  .section-note {font-size: 12px; margin-bottom: 12px;}
  .footer {font-size: 8px; color: #666; border-top: 1px solid #ccc; padding-top: 8px; margin-top: 16px;}
  .no-data {color: #c00; font-weight: bold; margin: 12px 0;}
</style>
</head>
<body>
<H3><U>EXPLICIT DATABASE LEVEL USERS – User Access Review Format</U></H3>

<div class="section-note">
<strong>IMPORTANT:</strong> Security group membership is managed outside SQL Server (Active Directory / Entra ID). 
Login Type is colored <span id="r">red</span> for accounts managed externally.<br><br>
<strong>Rows returned:</strong> ' + CAST(@RowCount AS varchar(10)) + '<br><br>
<strong>Copy/Paste Instructions:</strong> Select the entire table below (including the header row) and paste directly into the auditor template starting at the “Record Count” header row.  
Empty columns are left blank for the solution owner / reviewer.
</div>';

-- Explicit zero-row handling
IF @RowCount = 0
BEGIN
    SET @body += '<div class="no-data">No matching database-level permission records found after applying the standard exclusions.</div>';
END
ELSE
BEGIN
    SET @body += '
<table border="1" cellspacing="0" cellpadding="0">
<tr>
  <th>Record Count</th>
  <th>User ID</th>
  <th>User Name</th>
  <th>Assigned Roles / Permissions</th>
  <th>Last Login Date<br>(relevant for impact analysis for termed people)</th>
  <th>Account Status</th>
  <th>Account Type</th>
  <th>Password Management (Generic IDs Only)</th>
  <th>Generic Account Purpose (Generic IDs Only)</th>
  <th>Authorized Users (Generic IDs Only)</th>
  <th>Role Owner/Approver Name</th>
  <th>Access Appropriate?<br>Approve/Modify/Remove</th>
  <th>Reviewer Comments</th>
  <th>Remediation Ticket Number</th>
  <th>Impact Assessment<br>Document steps taken and document references if an impact assessment is necessary.</th>
</tr>';

    DECLARE @i INT = 0, @pos INT, @s VARCHAR(MAX) = '', @ts VARCHAR(MAX), @xml VARCHAR(MAX);

    -- Force @xml to never be NULL (critical when zero rows, but also safe when rows exist)
    SET @xml = ISNULL(CAST((
        SELECT
            ROW_NUMBER() OVER (ORDER BY LoginName, PermissionType) AS 'td', '',
            LoginName AS 'td', '',
            DatabaseUserName AS 'td', '',
            ('Database Level | Server: ' + ServerName + 
             ' | Database: ' + ISNULL(DatabaseName,'') + 
             ' | Role: ' + ISNULL([Role],'') + 
             ' | Permission: ' + ISNULL(PermissionType,'')) AS 'td', '',
            '' AS 'td', '',                                   -- Last Login Date
            AccountStatus AS 'td', '',                        -- Account Status
            CASE WHEN LoginType LIKE '%GROUP%' 
                 THEN 'zr' + CAST(LoginType AS VARCHAR(255))
                 ELSE CAST(LoginType AS VARCHAR(255))
            END AS 'td', '',
            CASE WHEN LoginName LIKE 'SVC[_]%' OR LoginName LIKE '%[_]SVC[_]%' OR LoginName LIKE 'svc%'
                 THEN 'Managed service account – see purpose column'
                 ELSE 'N/A for named users'
            END AS 'td', '',
            CASE WHEN LoginName LIKE 'SVC[_]%' OR LoginName LIKE '%[_]SVC[_]%' OR LoginName LIKE 'svc%'
                 THEN 'Service / application account (review purpose with owner)'
                 ELSE 'N/A for named users'
            END AS 'td', '',
            CASE WHEN LoginName LIKE 'SVC[_]%' OR LoginName LIKE '%[_]SVC[_]%' OR LoginName LIKE 'svc%'
                 THEN 'Confirm authorized users / process owners'
                 ELSE 'N/A for named users'
            END AS 'td', '',
            '' AS 'td', '',                                   -- Role Owner
            '' AS 'td', '',                                   -- Access Appropriate?
            '' AS 'td', '',                                   -- Reviewer Comments
            '' AS 'td', '',                                   -- Ticket
            '' AS 'td'                                        -- Impact Assessment
        FROM #AuditTemp3
        ORDER BY LoginName, PermissionType
        FOR XML PATH('tr'), ELEMENTS
    ) AS NVARCHAR(MAX)), '');

    SET @xml = REPLACE(@xml, '<td>zr', '<td id="r">');

    -- Zebra striping
    SELECT @s = '', @pos = CHARINDEX('<tr>', @xml, 4);
    WHILE (@pos > 0)
    BEGIN
        SET @i += 1;
        SET @ts = SUBSTRING(@xml, 1, @pos - 1);
        IF (@i % 2 = 1) SET @ts = REPLACE(@ts, '<tr>', '<tr id="odd">');
        SET @s += @ts;
        SET @xml = SUBSTRING(@xml, @pos, LEN(@xml));
        SET @pos = CHARINDEX('<tr>', @xml, 4);
    END
    SET @i += 1;
    SET @ts = @xml;
    IF (@i % 2 = 1) SET @ts = REPLACE(@ts, '<tr>', '<tr id="odd">');
    SET @s += @ts;

    SET @body += @s + '</table>';
END

-- Common footer
SET @body += '

Source table: bib_dba_utilities.dbo.permission_audit_temp<br>

Account Status sourced from sys.server_principals.is_disabled (Enabled / Disabled / N/A).
</div>

BIBERK Database Solutions Team
</body></html>';

DECLARE @MailSubject VARCHAR(255) = '[FOR INFORMATION / REVIEW] - Quarterly User Access Review EXPLICIT DATABASE LEVEL ' + @@SERVERNAME;

EXEC msdb.dbo.sp_send_dbmail
    @profile_name = 'SQLServer',
    @recipients   = 'stephen.mccord@BIBERK.com',   -- DEBUGGING
    @subject      = @MailSubject,
    @body         = @body,
    @body_format  = 'HTML';

DROP TABLE #AuditTemp3;
